Выдай ответ строго в формате JSON со следующими полями:

contents: [{ parts: [{ text: "Анализируй объект" }] }],
generationConfig: {
responseMimeType: "application/json",
responseSchema: {
type: "OBJECT",
properties: {
prosCons: { type: "STRING" },
pitch: { type: "STRING" },
socialPost: { type: "STRING" }
},
required: ["prosCons", "pitch", "socialPost"]
}

parts: [{ text: systemPrompt }]
}
};

method: 'POST',
headers: { 'Content-Type': 'application/json' },
body: JSON.stringify(payload)
